pub const CLIENT_API_VERSION: u32 = 1;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct NewAccount {
    pub account_id: AccountId,
    pub provider: Provider,
    pub label: Option<String>,
    pub config_dir: Option<String>,
}

#[derive(Clone, Debug, PartialEq, Eq, thiserror::Error)]
pub enum Error {
    InvalidLink { message: String },
    Pairing { message: String },
    UnknownMachine { host_id: HostId },
    Rejected { info: ErrorInfo },
    Local { message: String },
    Closed,
}

#[derive(Clone, Debug, PartialEq)]
pub struct Machine {
    pub host_id: HostId,
    pub name: String,
    pub addresses: Vec<String>,
    pub fingerprint: String,
    pub connection: ConnectionState,
    pub role: Option<Role>,
    pub sessions: Vec<SessionHead>,
    pub hosts: Vec<FleetHost>,
    pub projects: Vec<Project>,
    pub accounts: Vec<Account>,
    pub failover: FailoverSettings,
    pub terminals: Vec<Terminal>,
    pub resources: Option<HostResources>,
    pub session_usage: HashMap<SessionId, SessionUsage>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ConnectionState {
    Connecting,
    Connected,
    Disconnected { error: String },
}

#[derive(Clone, Debug, Default, PartialEq)]
pub struct SessionUpdate {
    pub events: Vec<Event>,
    pub streaming: Vec<Item>,
}
